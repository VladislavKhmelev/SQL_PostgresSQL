Создать временную таблицу с общей суммой completed заказов каждого клиента.

Затем:

определить среднюю сумму клиентов внутри каждого города;
оставить клиентов выше средней своего города;
среди оставшихся определить RANK() внутри города;
вывести клиентов с рангом не выше 3.

CREATE temp TABLE tt as (

    select
    id,
    name,

    COALESCE(sum(case when status = 'completed' then amount end),0) as r



    from customers
    LEFT JOIN orders
    on orders.customer_id = customers.id 

    GROUP BY id, name, city
);

with tt1 as (

    SELECT
    *,
    avg(r) over(PARTITION BY city) as r1

    from tt
),

tt2 as (
    SELECT
*

    from tt1
    where r>r1
),

tt3 as (
    SELECT
    *,
    RANK() OVER(PARTITION BY city ORDER BY r DESC) as r2

    from tt2
)

SELECT
*
from tt3
where r2 <= 3;
