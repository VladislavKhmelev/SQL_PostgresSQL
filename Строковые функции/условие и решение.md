Создать TEMP TABLE с:

id
customer_id
нормализованным order_code;
нормализованным status;
amount.

Затем через CASE определить категорию:

completed + сумма >= 5000 → 'premium'
completed + сумма < 5000 → 'normal'
cancelled → 'cancelled'.

Вывести только premium.

CREATE temp table tt (
    select
    id,
    customer_id,
    lower(trim(order_code)) as order_code,
    lower(trim(status)) as status,
    amount::numeric as amount
    from raw_orders
);


with tt1 as ( 
SELECT
*,
case when status = 'completed' and amount >= 5000 then 'premium'
when status = 'completed' and amount < 5000 then 'normal'
when status = 'cancelled' then 'cancelled' end as r

from tt
)

SELECT
*
from tt1
where r = 'premium';

Создать VIEW, в котором для каждого клиента:

очищенный email;
username;
domain.

Через CASE определить:

если email содержит @ и домен содержит . → 'valid';
иначе → 'invalid'.

CREATE VIEW tt as (
    SELECT
lower(trim(email)) as email,
split_part(lower(trim(email)),'@',1) as username,
split_part(lower(trim(email)),'@',2) as domain



    from raw_customers
);

SELECT
case when POSITION('@' in email) >0 and POSITION('.' in split_part(email, '@', 2))>0 then  'valid' else 'invalid' end r
from tt;

CREATE temp table tt as (

    select
    id, 
    initcap(trim(full_name)) as full_name,
    lower(trim(email)) as email,
    REGEXP_REPLACE(phone, '[^0-9]', '', 'g') as phone,
    initcap(trim(city)) as city,
    initcap(trim(country)) as country


    from raw_customers
);

SELECT
*
from tt 
where city = 'Moscow';
