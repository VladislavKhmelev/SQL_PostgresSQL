Задача

Найди сотрудников, которые одновременно:

получают больше средней зарплаты своего отдела;
работают в отделе, где минимум 2 сотрудника;
средняя зарплата этого отдела выше средней зарплаты по всей компании.


WHERE
↓
Subquery
↓
коррелированный Subquery
↓
CTE (WITH)
↓
GROUP BY
↓
HAVING
↓
JOIN
↓
IN
↓
EXISTS
↓
CTE + JOIN + агрегаты + Subquery



with tt as (
    select
    *
    from departments
    INNER JOIN employees
    on departments.id = employees.department_id
),

tt1 as (
    SELECT
    department,
    avg(salary) as r,
    count(*) as r1
    from tt
    GROUP BY department
    HAVING count(*) >= 2 and avg(salary) > (
        select
        avg(salary)
        from employees
    )
)
select 
name,
department
from tt
JOIN tt1
on tt.department = tt1.department
where tt.salary > tt1.r;
