определить его первый заказ;
определить его последний заказ;
посчитать разницу между ними;
посчитать общую сумму всех заказов;
посчитать среднее трёх последних заказов;
определить место клиента среди клиентов своего города по общей сумме;
вывести только клиентов, которые одновременно:
имеют минимум 3 заказа;
находятся в топ-3 своего города;
их последний заказ выше среднего трёх последних заказов.

with tt as (
    SELECT
    *,

    FIRST_VALUE(amount) over(PARTITION BY customer_id ORDER BY created_at) as r,

    FIRST_VALUE(amount) over(PARTITION BY customer_id ORDER BY created_at DESC) as r1,

    COALESCE(sum(amount) over(PARTITION BY customer_id),0) as r3,

    AVG(amount) OVER (
    PARTITION BY customer_id
    ORDER BY created_at DESC
    ROWS BETWEEN CURRENT ROW AND 2 FOLLOWING
) as r4,

    count(orders.id) over(PARTITION BY customer_id) as r5,

    ROW_NUMBER() OVER (
    PARTITION BY customer_id
    ORDER BY created_at DESC
) AS rn

    from customers
    LEFT JOIN orders
    on orders.customer_id = customers.id

 
),

tt1 as (
    select
    *,
    r1 - r as r2,

    DENSE_RANK() over(PARTITION BY city ORDER BY r3 DESC) as r6


    from tt
)

select
*

from tt1
where r5 >= 3 and r6 <=3 and r1 > r4 and rn = 1;

Найди сотрудников, которые одновременно:

получают больше средней зарплаты своего отдела;

with tt as (
    select
    *
    from departments
    INNER JOIN employees
    on departments.id = employees.department_id
),

tt1 as (
    SELECT
    department,
    avg(salary) as r,
    count(*) as r1
    from tt
    GROUP BY department
    HAVING count(*) >= 2 and avg(salary) > (
        select
        avg(salary)
        from employees
    )
)
select 
name,
department
from tt
JOIN tt1
on tt.department = tt1.department
where tt.salary > tt1.r;
