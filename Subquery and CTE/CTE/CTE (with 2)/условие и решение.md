Нужно найти отделы, где:

средняя зарплата сотрудников отдела выше 100 000;
сотрудников в отделе минимум 2.


with tt as (
    select 
    *
    from departments
    INNER JOIN employees
    on departments.id = employees.department_id
),

tt1 as (
    select
    department,
    avg(salary) as e,
    count(*) as e1
    from tt
    group by department
)

select 
department,
e,
e1
from tt1
where e > 100000 and e1 >= 2;

    *

    INNER JOIN employees

),

tt1 as (
    SELECT
    department,
    avg(salary) as r,
    count(*) as r1

department,
name
