Создать транзакцию, в которой:

баланс счёта id = 2 уменьшается на 1000;
создаётся операция withdrawal;
затем выясняется, что операция должна быть отменена.

Отменить все изменения через ROLLBACK.

BEGIN;

update accounts
SET balance = balance - 1000
where id = 2;


insert into account_operations
(account_id, operation_type, amount, created_at, status)
VALUES
(2, 'withdrawal', 1000, CURRENT_DATE, 'completed');

ROLLBACK;

(account_id, operation_type, amount, created_at, status)
