Затем:

через AVG() OVER() определить среднюю сумму всех клиентов;
через DENSE_RANK() определить место клиента среди всех;
через CASE определить:
ранг <= 3 → 'top';
сумма выше средней → 'above_average';
иначе → 'other'.

Вывести клиентов категории top или above_average.


CREATE temp table tt as (
    select 
id, 
name,

COALESCE(sum(case when status = 'completed' then amount end),0) as r




    from customers
LEFT JOIN orders
on orders.customer_id = customers.id  

    GROUP BY id, name
);

with tt1 as (

    SELECT
    *,
    avg(r) over() as r1,

    DENSE_RANK() over(ORDER BY r DESC) as r2

    from tt
),

tt2 as (

SELECT
*,
case when r2 <= 3 then 'top'
when r > r1 then 'above_average' else 'other' end as r3

from tt1
)

SELECT
*

from tt2
where r3 = 'top' or r3 = 'above_average';

r.name,
