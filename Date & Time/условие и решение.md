Для каждого месяца получить:

месяц;

максимальный completed;
место месяца по сумме completed;
сумму предыдущего месяца;
изменение суммы;

Категория:

strong:
сумма completed >= 10000
И completed >= 3

medium:
сумма completed >= 5000

weak:
иначе

Вывести только strong и medium.

Дополнительное условие: если у предыдущего месяца сумма равна 0, процентное изменение не должно выдавать ошибку деления на ноль.


CREATE temp table tt as (

with tt1 as (
    select
    *
    from customers
    INNER JOIN  orders
    on orders.customer_id = customers.id

    where extract(YEAR from created_at) = 2025
),

tt2 as (

    select
    date_trunc('month', created_at) as r,

    count(orders.id) as r1,

    count(case when status = 'completed' then 1 end) as r2,

    count(case when status = 'cancelled' then 1 end) as r3,

    sum(case when status = 'completed' then amount end) as r4,

    avg(case when status = 'completed' then amount end) as r5,

    max(case when status = 'completed' then amount end) as r6

    from tt1

    GROUP BY date_trunc('month', created_at)
),

tt3 as (
    select
    *,
    RANK() over(ORDER BY r4 DESC) as r7,

    lag(r4) over(ORDER BY r) as r8,

    r4 - lag(r4) over(ORDER BY r) as r9,

    (r4 -  COALESCE(lag(r4) over(ORDER BY r),0))::numeric / NULLIF(lag(r4) over(ORDER BY r),0) * 100 as r10,


    case when r4 >= 10000 and r2 >= 3 then 'strong'
    when r4 >= 5000 then 'medium' else 'weak' end f

    from tt2
)

select
*

from tt3
where f = 'strong' or f = 'medium'

);
