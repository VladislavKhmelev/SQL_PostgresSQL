with tt as (
    SELECT
    *,
    FIRST_value(amount) over(PARTITION BY customer_id ORDER BY date) as r
    from sales
)

select 
*

сравнить продажу с первой продажей

Нужно для каждой строки показать:

customer_id
date
amount
first_amount
difference

Где:

first_amount = первая продажа клиента
difference = amount - first_amount



with tt as (
    SELECT
    *,
    FIRST_value(amount) over(PARTITION BY customer_id ORDER BY date) as r
    from sales
)

select 
*,
amount - r as r1
from tt;
