Задание

Найди сотрудников, которые получают максимальную зарплату в своём отделе.

| id | department |
| -: | ---------- |
|  1 | IT         |
|  2 | Sales      |
|  3 | HR         |
|  4 | Marketing  |

| id | name    | department_id | salary |
| -: | ------- | ------------: | -----: |
|  1 | Иван    |             1 | 120000 |
|  2 | Анна    |             1 | 140000 |
|  3 | Пётр    |             2 |  90000 |
|  4 | Мария   |             2 | 130000 |
|  5 | Сергей  |             2 | 110000 |
|  6 | Олег    |             3 |  70000 |
|  7 | Елена   |             4 | 150000 |
|  8 | Дмитрий |             4 | 130000 |



with tt as (
    select 
    *
    from departments
    INNER JOIN employees
    on departments.id = employees.department_id
),

tt1 as (
    SELECT
    department,
    max(salary) as e
    from tt
    GROUP BY department
)

SELECT
*
from tt 
join tt1
on tt.department = tt1.department
where tt.salary = tt1.e;
