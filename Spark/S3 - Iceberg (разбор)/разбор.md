# ────────────────────────────────────────────
# POSTGRESQL
# ────────────────────────────────────────────

load_dotenv(
    r"C:\Users\VladK\OneDrive\Desktop\proj\ver10\1.env"

pg_host = os.getenv("DB_HOST")

# ────────────────────────────────────────────
# CLOUD.RU S3
# ────────────────────────────────────────────

load_dotenv(
    r"C:\Users\VladK\OneDrive\Desktop\proj\ver10\cloud_s3.env"
)

s3_endpoint = os.getenv("S3_ENDPOINT")

spark = (
    SparkSession.builder

    .appName("SparkApp")

    # PostgreSQL JDBC Driver
    .config(
        "spark.jars",
        r"C:\Users\VladK\OneDrive\Desktop\proj\ver10\jars\PostgreSQL JDBC Driver\postgresql-42.7.12.jar"
    )

    # Hadoop AWS + Iceberg
    .config(
        "spark.jars.packages",

df_customers = spark.read.csv(

)

    inferSchema=True
)

df_customers.writeTo(
    "iceberg.raw.customers"
).createOrReplace()

    "iceberg.raw.orders"
).createOrReplace()

    "iceberg.raw.products"
).createOrReplace()








df_customers_iceberg = spark.sql(""" 
    SELECT *
    FROM iceberg.raw.customers
""")

df_customers_iceberg.show()






df_valid_customers = spark.sql("""

SELECT
customer_id,
first_name,
last_name,
email,
city,
created_at,

case when customer_id is null then 'customer_id is null'

when first_name is null then 'first_name is null'

when last_name is null then 'last_name is null'

when email is null then 'email is null'

when city is null then 'city is null'

when created_at is null then 'created_at is null'
end as error_reason

from iceberg.raw.customers;

""")


df_valid_customers.writeTo(
    "iceberg.validated.customers"
).createOrReplace()






df_customer_core = spark.sql("""
SELECT
customer_id,
first_name,
last_name,
email,
city,
created_at


from iceberg.validated.customers 

where error_reason is null;

""")


df_customer_core.writeTo(
    "iceberg.core.customers"
).createOrReplace()




df_customers_update = spark.createDataFrame([
    (1, "Ivan", "Petrov", "ivan_new@mail.com", "Ufa", "2026-10-01 10:00:00"),
    (6, "Sergey", "Orlov", "sergey@mail.com", "Perm", "2026-10-01 11:00:00")
], [
    "customer_id",
    "first_name",
    "last_name",
    "email",
    "city",
    "created_at"
])

df_customers_update = df_customers_update.withColumn(
    "created_at",
    df_customers_update["created_at"].cast("timestamp")
)

df_customers_update.createOrReplaceTempView("df_customers_update")


spark.sql("""
    SELECT *
    FROM df_customers_update 
""").show()









dd = spark.sql("""
MERGE INTO iceberg.core.customers AS t
USING df_customers_update AS s
ON t.customer_id = s.customer_id

WHEN MATCHED THEN
UPDATE SET
    t.first_name = s.first_name,
    t.last_name = s.last_name,
    t.email = s.email,
    t.city = s.city,
    t.created_at = s.created_at

WHEN NOT MATCHED THEN
INSERT (
    customer_id,  
    first_name,
    last_name,
    email,
    city,
    created_at
)
VALUES (
    s.customer_id,
    s.first_name,
    s.last_name,
    s.email,
    s.city,
    s.created_at
)
""")



dd.show()
